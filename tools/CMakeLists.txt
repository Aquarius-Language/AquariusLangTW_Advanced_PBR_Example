cmake_minimum_required(VERSION 3.20)
project(PbrAssetBaker LANGUAGES C CXX)
set(PBR_SOURCE "C:/GitClonedProjects/PBR" CACHE PATH "Original Nadrin/PBR checkout")
set_source_files_properties("${PBR_SOURCE}/lib/glad/src/glad.c" "${PBR_SOURCE}/lib/stb/src/libstb.c" PROPERTIES LANGUAGE CXX)
add_executable(bake bake.cpp "${PBR_SOURCE}/src/opengl.cpp" "${PBR_SOURCE}/src/common/mesh.cpp" "${PBR_SOURCE}/src/common/image.cpp" "${PBR_SOURCE}/src/common/utils.cpp" "${PBR_SOURCE}/lib/glad/src/glad.c" "${PBR_SOURCE}/lib/stb/src/libstb.c")
target_compile_features(bake PRIVATE cxx_std_17)
target_compile_options(bake PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/FIstdexcept>)
target_compile_definitions(bake PRIVATE ENABLE_OPENGL GLFW_INCLUDE_NONE GLM_ENABLE_EXPERIMENTAL NOMINMAX)
target_include_directories(bake PRIVATE "${PBR_SOURCE}/src" "${PBR_SOURCE}/lib/glad/include" "${PBR_SOURCE}/lib/glm/include" "${PBR_SOURCE}/lib/stb/include" "${PBR_SOURCE}/lib/assimp/include" "${PBR_SOURCE}/lib/glfw/include")
target_link_libraries(bake PRIVATE "${PBR_SOURCE}/lib/assimp/win64/assimp.lib" "${PBR_SOURCE}/lib/glfw/win64/glfw3dll.lib" opengl32)
add_custom_command(TARGET bake POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PBR_SOURCE}/lib/assimp/win64/assimp.dll" "${PBR_SOURCE}/lib/glfw/win64/glfw3.dll" "$<TARGET_FILE_DIR:bake>")
